#include "zf_common_headfile.h"
#include "uart1.hpp"
#include <unistd.h>

int main()
{
    uart1_init();

    uint32 count = 0;
    while (1)
    {
        // 发送 8 字节数据: 帧头 0xAA 0x55 + uint32 count + uint16 校验和
        uint8 buf[8];
        buf[0] = 0xAA;
        buf[1] = 0x55;
        buf[2] = (count >> 0)  & 0xFF;
        buf[3] = (count >> 8)  & 0xFF;
        buf[4] = (count >> 16) & 0xFF;
        buf[5] = (count >> 24) & 0xFF;

        uint16 checksum = 0xAA + 0x55
                        + buf[2] + buf[3] + buf[4] + buf[5];
        buf[6] = (checksum >> 0) & 0xFF;
        buf[7] = (checksum >> 8) & 0xFF;

        uart1_send(buf, 8);
        count++;

        usleep(3000000); // 2s
    }

    uart1_close();
    return 0;
}
